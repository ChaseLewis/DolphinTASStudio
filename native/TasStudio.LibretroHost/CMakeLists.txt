cmake_minimum_required(VERSION 3.25)
project(TasStudioLibretroHost LANGUAGES CXX)
add_executable(tas-native-smoke smoke.cpp)
target_compile_features(tas-native-smoke PRIVATE cxx_std_20)
target_compile_definitions(tas-native-smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(tas-native-smoke PRIVATE /W4 /EHsc)
endif()
add_library(TasStudio.LibretroHost SHARED host.cpp)
target_compile_features(TasStudio.LibretroHost PRIVATE cxx_std_20)
target_compile_definitions(TasStudio.LibretroHost PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(TasStudio.LibretroHost PRIVATE ../dolphin-libretro/Externals/libretro-common/include)
target_link_libraries(TasStudio.LibretroHost PRIVATE d3d11 d3dcompiler)
if(MSVC)
  target_compile_options(TasStudio.LibretroHost PRIVATE /W4 /EHsc)
endif()
