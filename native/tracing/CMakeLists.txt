cmake_minimum_required(VERSION 3.25)
project(TasTraceChecks LANGUAGES CXX)
enable_testing()
add_executable(trace-buffer-tests buffer-tests.cpp)
add_executable(trace-smoke trace-smoke.cpp)
foreach(target trace-buffer-tests trace-smoke)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  endif()
endforeach()
add_test(NAME trace-buffer COMMAND trace-buffer-tests)
